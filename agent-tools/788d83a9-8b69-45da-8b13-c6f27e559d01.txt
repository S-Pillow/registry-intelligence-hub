 }

 location: items.first[:location],
 count: items.count,
 domains: items.map{|x| x[:domain]}.join(", "),

 -(x[:count])

 puts "[No.#{i+1}] #{x[:contact]} (#{x[:location]})"
 puts "#{x[:count]} (#{x[:domains]})"

end
